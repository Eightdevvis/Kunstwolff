---
type Slide = {
  src: string;
  alt?: string;
};

interface Props {
  items?: Slide[];
}

const { items = [] } = Astro.props;
---

<section class="section gallery" id="work">
  <div class="section-inner">
    <h2>Unsere Kunst</h2>

    <div class="slider-wrapper">
      <button class="nav prev" aria-label="Zurück">‹</button>

      <div class="gallery-swipe" id="slider">
        {items.map((slide) => (
          <div class="slide">
            <img
              src={slide.src}
              alt={slide.alt ?? ''}
              loading="lazy"
              draggable="false"
            />
          </div>
        ))}
      </div>

      <button class="nav next" aria-label="Weiter">›</button>
    </div>
  </div>
</section>

<script>
  const slider = document.getElementById('slider');
  const prev = document.querySelector('.prev');
  const next = document.querySelector('.next');

  if (!slider) return;

  const getSlideWidth = () => {
    const slide = slider.querySelector('.slide');
    return slide ? slide.offsetWidth : 0;
  };

  prev?.addEventListener('click', () => {
    slider.scrollBy({
      left: -getSlideWidth(),
      behavior: 'smooth',
    });
  });

  next?.addEventListener('click', () => {
    slider.scrollBy({
      left: getSlideWidth(),
      behavior: 'smooth',
    });
  });

  setInterval(() => {
    const maxScroll = slider.scrollWidth - slider.clientWidth;

    if (slider.scrollLeft >= maxScroll - 10) {
      slider.scrollTo({ left: 0, behavior: 'smooth' });
    } else {
      slider.scrollBy({ left: getSlideWidth(), behavior: 'smooth' });
    }
  }, 4000);
</script>
